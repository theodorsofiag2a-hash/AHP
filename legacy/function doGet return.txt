      .addMetaTag('viewport', 'width=device-width, initial-scale=1');

function initDados() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetModelos = ss.getSheetByName("SYS_Modelos");
  if (!sheetModelos) {
    sheetModelos = ss.insertSheet("SYS_Modelos");
    sheetModelos.appendRow(["ID", "Nome do Modelo", "Estrutura JSON"]);
  }
  return getModelos();
}

function getModelos() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("SYS_Modelos");
  if (!sheet) return [];
  var ultimaLinha = sheet.getLastRow();
  if (ultimaLinha < 2) return [];
  var dados = sheet.getRange(2, 1, ultimaLinha - 1, 3).getValues();
  var lista = [];
  for (var i = 0; i < dados.length; i++) {
    lista.push({ id: dados[i][0], nome: dados[i][1], estrutura: JSON.parse(dados[i][2]) });
  }
  return lista;
}

function salvarModelo(modelo) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("SYS_Modelos");
  var dados = sheet.getDataRange().getValues();
  var atualizado = false;

  for (var i = 1; i < dados.length; i++) {
    if (dados[i][1] === modelo.nome) {
      sheet.getRange(i + 1, 3).setValue(JSON.stringify(modelo.estrutura));
      atualizado = true; break;
    }
  }

  if (!atualizado) {
    var novoId = sheet.getLastRow();
    sheet.appendRow([novoId, modelo.nome, JSON.stringify(modelo.estrutura)]);
  }
  
  var nomeAba = "BD_" + modelo.nome.replace(/[^a-zA-Z0-9]/g, "_").substring(0, 20);
  if (!ss.getSheetByName(nomeAba)) {
    var newSheet = ss.insertSheet(nomeAba);
    newSheet.appendRow(["ID", "Nome Avaliado", "Avaliacao JSON"]);
  }
  return { sucesso: "Modelo '" + modelo.nome + "' guardado com sucesso!" };
}

function excluirModeloDatabase(nomeModelo) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("SYS_Modelos");
  var dados = sheet.getDataRange().getValues();

  for (var i = 1; i < dados.length; i++) {
    if (dados[i][1] === nomeModelo) { sheet.deleteRow(i + 1); break; }
  }

  var nomeAba = "BD_" + nomeModelo.replace(/[^a-zA-Z0-9]/g, "_").substring(0, 20);
  var sheetBD = ss.getSheetByName(nomeAba);
  if (sheetBD) ss.deleteSheet(sheetBD);

  return { sucesso: "Modelo excluído permanentemente." };
}

function salvarAvaliacao(nomeModelo, avaliacao) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var nomeAba = "BD_" + nomeModelo.replace(/[^a-zA-Z0-9]/g, "_").substring(0, 20);
  var sheet = ss.getSheetByName(nomeAba);
  if(!sheet) return {erro: "Base de dados não encontrada."};
  
  var dados = sheet.getDataRange().getValues();
  var atualizado = false;
  
  // Lógica de Edição (Update)
  for(var i = 1; i < dados.length; i++) {
    if(dados[i][1] === avaliacao.nome) {
      sheet.getRange(i + 1, 3).setValue(JSON.stringify(avaliacao.notas));
      atualizado = true; break;
    }
  }
  
  // Lógica de Inserção (Create)
  if(!atualizado) {
    var novoId = sheet.getLastRow();
    sheet.appendRow([novoId, avaliacao.nome, JSON.stringify(avaliacao.notas)]);
  }
  
  return { sucesso: "Ficha de " + avaliacao.nome + " gravada com sucesso!" };
}

function excluirAvaliacao(nomeModelo, nomeEmpresa) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var nomeAba = "BD_" + nomeModelo.replace(/[^a-zA-Z0-9]/g, "_").substring(0, 20);
  var sheet = ss.getSheetByName(nomeAba);
  if(!sheet) return {erro: "Base de dados não encontrada."};
  
  var dados = sheet.getDataRange().getValues();
  for(var i = 1; i < dados.length; i++) {
    if(dados[i][1] === nomeEmpresa) {
      sheet.deleteRow(i + 1);
      return { sucesso: "Empresa " + nomeEmpresa + " excluída." };
    }
  }
  return { erro: "Empresa não encontrada." };
}

// ================= CÁLCULO AHP =================
function calcularAHPGenerico(nomeModelo, estrutura, comparacoesArray) {
  var n = estrutura.length;
  var matriz = [];
  for(var i=0; i<n; i++) matriz[i] = new Array(n).fill(1);
  
  var k = 0;
  for(var i=0; i<n; i++){
    for(var j=i+1; j<n; j++){
      var val = parseFloat(comparacoesArray[k]);
      matriz[i][j] = val; matriz[j][i] = 1/val;
      k++;
    }
  }
  
  var somaColunas = new Array(n).fill(0);
  for (var i = 0; i < n; i++) {
    for (var j = 0; j < n; j++) somaColunas[j] += matriz[i][j];
  }
  
  var pesosPilares = [];
  for (var i = 0; i < n; i++) {
    var somaLinha = 0;
    for (var j = 0; j < n; j++) somaLinha += matriz[i][j] / somaColunas[j];
    pesosPilares.push(somaLinha / n);
  }
  
  var lambdaMax = 0;
  for (var i = 0; i < n; i++) lambdaMax += pesosPilares[i] * somaColunas[i];
  var RI_DICT = {1:0, 2:0, 3:0.58, 4:0.9, 5:1.12, 6:1.24, 7:1.32, 8:1.41};
  var RI = RI_DICT[n] || 1.45;
  var CR = n > 2 ? ((lambdaMax - n) / (n - 1)) / RI : 0;
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var nomeAba = "BD_" + nomeModelo.replace(/[^a-zA-Z0-9]/g, "_").substring(0, 20);
  var sheet = ss.getSheetByName(nomeAba);
  if(!sheet || sheet.getLastRow() < 2) return {erro: "Nenhum fornecedor foi avaliado ainda neste modelo."};
  
  var dados = sheet.getRange(2, 1, sheet.getLastRow()-1, 3).getValues();
  var ranking = [];
  
  for(var f=0; f<dados.length; f++){
    var nomeAvaliado = dados[f][1];
    var notasDict = JSON.parse(dados[f][2]); 
    
    var scoreFinal = 0; var pScores = []; var pScoresRadar = []; 
    
    for(var i=0; i<n; i++){
      var pilar = estrutura[i].pilar;
      var subs = estrutura[i].subcriterios;
      
      var somaPesosSubs = 0;
      for(var j=0; j<subs.length; j++) somaPesosSubs += subs[j].peso;
      if(somaPesosSubs === 0) somaPesosSubs = 1; 
      
      var desempenhoPilar = 0;
      for(var j=0; j<subs.length; j++){
         var subNome = subs[j].nome;
         var pesoSubRelativo = subs[j].peso / somaPesosSubs;
         var desempenhoEscala = notasDict[pilar] && (notasDict[pilar][subNome] !== undefined) ? notasDict[pilar][subNome] : 0;
         desempenhoPilar += (desempenhoEscala / 100) * pesoSubRelativo;
      }
      
      var notaPonderada = desempenhoPilar * pesosPilares[i] * 1000;
      scoreFinal += notaPonderada; pScores.push(notaPonderada); pScoresRadar.push(desempenhoPilar * 10);
    }
    
    ranking.push({ 
      nome: nomeAvaliado, nota: scoreFinal.toFixed(0), 
      pScores: pScores.map(v => parseFloat(v).toFixed(0)), pScoresRadar: pScoresRadar.map(v => parseFloat(v).toFixed(1))
    });
  }
  
  ranking.sort((a,b) => b.nota - a.nota);
  return { cr: (CR * 100).toFixed(2), pesos: pesosPilares, ranking: ranking, estrutura: estrutura };
}
